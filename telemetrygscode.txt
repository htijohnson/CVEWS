// Returns the sheet for a given Date (defaults to now), creating it with headers if missing
function getMonthlySheet(ss, targetDate) {
  var dateObj = targetDate || new Date();
  var tz = Session.getScriptTimeZone();
  var sheetName = Utilities.formatDate(dateObj, tz, "yyyy-MM");
  
  var sheet = ss.getSheetByName(sheetName);
  if (!sheet) {
    sheet = ss.insertSheet(sheetName);
    sheet.appendRow(LOG_HEADERS);
    sheet.getRange(1, 1, 1, LOG_HEADERS.length).setFontWeight("bold");
    sheet.setFrozenRows(1);
  }
  return sheet;
}

// Fetches recent rows across the current and previous month's sheets seamlessly
function getRecentTelemetryRows(ss, maxRows) {
  var now = new Date();
  var currSheet = getMonthlySheet(ss, now);
  var rows = [];

  var currLast = currSheet.getLastRow();
  if (currLast >= 2) {
    var takeCurr = Math.min(currLast - 1, maxRows);
    rows = currSheet.getRange(currLast - takeCurr + 1, 1, takeCurr, 19).getValues();
  }

  // Pull remainder from previous month's tab if early in a new month
  if (rows.length < maxRows) {
    var prevMonthDate = new Date(now.getFullYear(), now.getMonth() - 1, 15);
    var tz = Session.getScriptTimeZone();
    var prevSheetName = Utilities.formatDate(prevMonthDate, tz, "yyyy-MM");
    var prevSheet = ss.getSheetByName(prevSheetName);

    if (prevSheet && prevSheet.getLastRow() >= 2) {
      var needed = maxRows - rows.length;
      var prevLast = prevSheet.getLastRow();
      var takePrev = Math.min(prevLast - 1, needed);
      var prevRows = prevSheet.getRange(prevLast - takePrev + 1, 1, takePrev, 19).getValues();
      rows = prevRows.concat(rows);
    }
  }
  return rows;
}

// Helper to format any standard US phone string into E.164 (+1XXXXXXXXXX)
function formatToE164(phone) {
  if (!phone) return null;
  var digits = String(phone).replace(/\D/g, "");
  if (digits.length === 10) {
    return "+1" + digits;
  } else if (digits.length === 11 && digits.charAt(0) === "1") {
    return "+" + digits;
  } else if (digits.length > 10) {
    return "+" + digits;
  }
  return null;
}

// Helper to escape XML characters for Twilio TwiML responses
function escapeXml(unsafe) {
  return String(unsafe || "").replace(/[<>&'"]/g, function (c) {
    switch (c) {
      case '<': return '&lt;';
      case '>': return '&gt;';
      case '&': return '&amp;';
      case '\'': return '&apos;';
      case '"': return '&quot;';
    }
  });
}

// Validates that PIN matches the last 4 of Column 3 (Phone) AND
// entered name matches (or is a part of) Column 2 (Full Name) on that same row
function validateRecipientCredentials(enteredName, enteredPin) {
  var cleanPin = String(enteredPin || "").trim();
  var cleanName = String(enteredName || "").trim().toLowerCase();

  if (!cleanPin || !cleanName) return false;

  // Master/Admin PIN override check
  if (cleanPin === String(ALARM_ACK_PIN).trim()) {
    return true;
  }

  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var recipientSheet = ss.getSheetByName("Recipients");
  if (!recipientSheet) return false;

  var lastRow = recipientSheet.getLastRow();
  if (lastRow < 2) return false;

  // Read Column 2 (Full Name) and Column 3 (Phone / Target)
  var data = recipientSheet.getRange(2, 2, lastRow - 1, 2).getValues();

  for (var i = 0; i < data.length; i++) {
    var sheetName = String(data[i][0] || "").trim().toLowerCase();
    var rawPhone = String(data[i][1] || "").replace(/\D/g, "");

    if (rawPhone.length >= 4) {
      var lastFour = rawPhone.slice(-4);

      if (cleanPin === lastFour) {
        if (sheetName.indexOf(cleanName) !== -1 || cleanName.indexOf(sheetName) !== -1) {
          return true;
        }
      }
    }
  }

  return false;
}

function sendTwilioSms(toPhone, bodyText) {
  var formattedTo = formatToE164(toPhone);
  if (!formattedTo) {
    Logger.log("Skipping SMS dispatch: Invalid phone number format: " + toPhone);
    return;
  }

  // Append snooze prompt to every outbound text if not already present
  var finalBody = String(bodyText || "");
  if (finalBody.toLowerCase().indexOf("respond with snooze") === -1) {
    finalBody += SNOOZE_FOOTER;
  }

  var url = "https://api.twilio.com/2010-04-01/Accounts/" + TWILIO_ACCOUNT_SID + "/Messages.json";
  var options = {
    "method": "post",
    "headers": {
      "Authorization": "Basic " + Utilities.base64Encode(TWILIO_ACCOUNT_SID + ":" + TWILIO_AUTH_TOKEN)
    },
    "payload": {
      "To": formattedTo,
      "From": TWILIO_FROM_NUMBER,
      "Body": finalBody
    },
    "muteHttpExceptions": true
  };
  try {
    var response = UrlFetchApp.fetch(url, options);
    var code = response.getResponseCode();
    var text = response.getContentText();
    Logger.log("Twilio HTTP " + code + " to " + formattedTo + ": " + text);
  } catch(e) {
    Logger.log("Error dispatching SMS to " + formattedTo + ": " + e.toString());
  }
}

function dispatchSmsToRecipients(messageBody) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var recipientSheet = ss.getSheetByName("Recipients");
  if (!recipientSheet) {
    Logger.log("Recipients sheet missing. SMS aborted.");
    return;
  }
  
  var lastRow = recipientSheet.getLastRow();
  if (lastRow < 2) return;
  
  // Read Columns A through F (Col C = Phone, Col F = Snoozed Until)
  var data = recipientSheet.getRange(2, 1, lastRow - 1, 6).getValues();
  var now = new Date();
  
  for (var i = 0; i < data.length; i++) {
    var phone = String(data[i][2]).trim();
    if (!phone) continue;

    // Check if recipient is currently snoozed (Column F / index 5)
    var snoozedUntilRaw = data[i][5];
    if (snoozedUntilRaw) {
      var snoozedUntil = new Date(snoozedUntilRaw);
      if (!isNaN(snoozedUntil.getTime()) && now < snoozedUntil) {
        Logger.log("Skipping SMS to " + phone + " (snoozed until " + snoozedUntil.toISOString() + ")");
        continue;
      }
    }

    sendTwilioSms(phone, messageBody);
  }
}

// INCOMING TWILIO SMS WEBHOOK HANDLER
// ==========================================
function handleIncomingTwilioSms(ss, fromNumber, messageBody) {
  var formattedFrom = formatToE164(fromNumber);
  var cleanBody = String(messageBody || "").trim().toUpperCase();
  var replyMessage = "";

  var recipientSheet = ss.getSheetByName("Recipients");
  if (!recipientSheet || recipientSheet.getLastRow() < 2 || !formattedFrom) {
    replyMessage = "Your number was not found on the CVEWS alert list.";
    return ContentService.createTextOutput(
      '<?xml version="1.0" encoding="UTF-8"?><Response><Message>' + escapeXml(replyMessage) + '</Message></Response>'
    ).setMimeType(ContentService.MimeType.XML);
  }

  var lastRow = recipientSheet.getLastRow();
  var phoneValues = recipientSheet.getRange(2, 3, lastRow - 1, 1).getValues();
  var matchedRows = [];

  for (var i = 0; i < phoneValues.length; i++) {
    if (formatToE164(phoneValues[i][0]) === formattedFrom) {
      matchedRows.push(i + 2); // 1-indexed sheet row
    }
  }

  if (matchedRows.length === 0) {
    replyMessage = "Your number was not found on the CVEWS alert list.";
  } else if (cleanBody === "SNOOZE" || cleanBody.indexOf("SNOOZE") === 0) {
    var snoozeUntil = new Date(Date.now() + (SNOOZE_DURATION_MINUTES * 60 * 1000));
    var tz = Session.getScriptTimeZone();
    var formattedResumeTime = Utilities.formatDate(snoozeUntil, tz, "h:mm a");

    // Ensure Column F header exists
    if (!recipientSheet.getRange(1, 6).getValue()) {
      recipientSheet.getRange(1, 6).setValue("Snoozed Until").setFontWeight("bold");
    }

    matchedRows.forEach(function(rowIdx) {
      recipientSheet.getRange(rowIdx, 6).setValue(snoozeUntil);
    });
    SpreadsheetApp.flush();

    replyMessage = "CVEWS alerts paused for " + SNOOZE_DURATION_MINUTES + " minutes (until " + formattedResumeTime + "). Reply RESUME anytime to turn them back on.";
  } else if (cleanBody === "RESUME" || cleanBody === "UNPAUSE" || cleanBody === "UNSNOOZE") {
    matchedRows.forEach(function(rowIdx) {
      recipientSheet.getRange(rowIdx, 6).clearContent();
    });
    SpreadsheetApp.flush();

    replyMessage = "CVEWS alerts have been resumed.";
  } else {
    replyMessage = "Unrecognized command. Reply SNOOZE to pause alerts for 30 minutes, RESUME to unpause, or STOP to unsubscribe.";
  }

  var twiml = '<?xml version="1.0" encoding="UTF-8"?><Response><Message>' + escapeXml(replyMessage) + '</Message></Response>';
  return ContentService.createTextOutput(twiml).setMimeType(ContentService.MimeType.XML);
}

// OPT-IN RECORD WRITER (USED BY GET & POST)
// ==========================================
function saveOptInRecord(ss, name, phone, ip, consentText) {
  var recipientSheet = ss.getSheetByName("Recipients");
  if (!recipientSheet) {
    recipientSheet = ss.insertSheet("Recipients");
  }
  
  if (recipientSheet.getLastRow() === 0) {
    recipientSheet.appendRow(["Timestamp", "Full Name", "Phone / Target", "IP Address", "Consent Disclosure", "Snoozed Until"]);
    recipientSheet.getRange("A1:F1").setFontWeight("bold");
  }

  var formattedPhone = formatToE164(phone) || phone;
  recipientSheet.appendRow([new Date(), name, formattedPhone, ip, consentText, ""]);
  SpreadsheetApp.flush();
  Logger.log("Successfully recorded opt-in for: " + name + " (" + formattedPhone + ")");
}

// 1. DATA LOGGER & FORM INGESTION ENTRY POINT
// ==========================================
function doPost(e) {
  var mLock = LockService.getScriptLock();
  try {
    mLock.waitLock(30000);
    
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var rawContents = (e && e.postData && e.postData.contents) ? e.postData.contents : "";
    Logger.log("Incoming doPost rawContents: " + rawContents);

    // ----------------------------------------------------
    // CHECK 0: Incoming Twilio SMS Webhook (From & Body)
    // ----------------------------------------------------
    if (e && e.parameter && e.parameter.From && typeof e.parameter.Body !== "undefined") {
      return handleIncomingTwilioSms(ss, e.parameter.From, e.parameter.Body);
    }

    var parsedJson = null;
    if (rawContents) {
      try {
        parsedJson = JSON.parse(rawContents);
      } catch (parseIgnore) {}
    }

    // ----------------------------------------------------
    // CHECK 1: Opt-in Form Submission
    // ----------------------------------------------------
    var isOptIn = false;
    var name = "", phone = "", ip = "Unknown", consentText = "";

    if (parsedJson && (parsedJson.type === "optin" || parsedJson.phone || parsedJson.consent_text)) {
      isOptIn = true;
      name = parsedJson.name || "";
      phone = parsedJson.phone || "";
      ip = parsedJson.ip || "Unknown";
      consentText = parsedJson.consent_text || "";
    } else if (e && e.parameter && (e.parameter.type === "optin" || e.parameter.phone || e.parameter.name)) {
      isOptIn = true;
      name = e.parameter.name || "";
      phone = e.parameter.phone || "";
      ip = e.parameter.ip || "Unknown";
      consentText = e.parameter.consent_text || "";
    } else if (rawContents && (rawContents.indexOf("type=optin") !== -1 || rawContents.indexOf("phone=") !== -1)) {
      isOptIn = true;
      var params = {};
      rawContents.split("&").forEach(function(part) {
        var item = part.split("=");
        if (item.length === 2) {
          params[decodeURIComponent(item[0])] = decodeURIComponent(item[1].replace(/\+/g, " "));
        }
      });
      name = params.name || "";
      phone = params.phone || "";
      ip = params.ip || "Unknown";
      consentText = params.consent_text || "";
    }

    if (isOptIn) {
      saveOptInRecord(ss, name, phone, ip, consentText);
      return ContentService.createTextOutput(JSON.stringify({ "status": "success", "message": "Opt-in saved successfully" }))
                           .setMimeType(ContentService.MimeType.JSON);
    }

    // ----------------------------------------------------
    // CHECK 2: ESP32 JSON Payload Ingestion
    // ----------------------------------------------------
    var payloadData = parsedJson;
    if (!payloadData) {
      Logger.log("Invalid or unrecognized data: " + rawContents);
      return ContentService.createTextOutput(JSON.stringify({"status": "error", "message": "Invalid JSON or unrecognized form data"}))
                           .setMimeType(ContentService.MimeType.JSON);
    }
    
    // Check for Alarm SMS Trigger (> 30s)
    if (payloadData.action === "alarm_sms") {
      var alertText = payloadData.message || "🚨 CVEWS CRITICAL ALERT: System Alarm Light active > 30s! Check https://htijohnson.github.io/CVEWS/";
      dispatchSmsToRecipients(alertText);
      return ContentService.createTextOutput(JSON.stringify({"status": "alarm_sms_dispatched"}))
                           .setMimeType(ContentService.MimeType.JSON);
    }

    // Check for Alarm Resolved SMS Trigger
    if (payloadData.action === "alarm_clear") {
      var clearText = payloadData.message || "✅ CVEWS RESOLVED: The System Alarm has cleared and returned to normal status.";
      dispatchSmsToRecipients(clearText);
      return ContentService.createTextOutput(JSON.stringify({"status": "alarm_clear_dispatched"}))
                           .setMimeType(ContentService.MimeType.JSON);
    }

    // Check for Daily Noon Test SMS (Deduplicated to 1 per calendar day across all nodes/reboots)
    if (payloadData.action === "daily_test_sms") {
      var tz = Session.getScriptTimeZone();
      var todayKey = Utilities.formatDate(new Date(), tz, "yyyy-MM-dd");
      var lastSentDate = scriptProps.getProperty("LAST_NOON_TEST_DATE");

      if (lastSentDate === todayKey) {
        Logger.log("Daily noon test SMS already dispatched today (" + todayKey + "). Skipping duplicate.");
        return ContentService.createTextOutput(JSON.stringify({"status": "skipped_already_sent_today"}))
                             .setMimeType(ContentService.MimeType.JSON);
      }

      // Lock in today's date BEFORE sending so concurrent/rapid requests cannot duplicate
      scriptProps.setProperty("LAST_NOON_TEST_DATE", todayKey);

      var histData = getRecentTelemetryRows(ss, 3000); // ~2880 rows per 24h across month boundary
      var snapshotLines = [];
      
      if (histData.length > 0) {
        var rData = histData[histData.length - 1];
        
        var rowTime = Utilities.formatDate(new Date(rData[0]), tz, "MM/dd HH:mm:ss");
        var tkSm    = Math.round(Number(rData[1]) || 0);
        var tkLg    = Math.round(Number(rData[2]) || 0);
        var flow    = (Number(rData[3]) || 0).toFixed(1);
        var cl2     = (Number(rData[4]) || 0).toFixed(2);
        var alm     = (rData[11] == 1 || rData[11] === "TRIPPED") ? "TRIPPED" : "CLEAR";

        var utHi    = (rData[15] == 1) ? "ALARM" : "OK";
        var utLo    = (rData[16] == 1) ? "ALARM" : "OK";
        var ltHi    = (rData[17] == 1) ? "ALARM" : "OK";
        var ltLo    = (rData[18] == 1) ? "ALARM" : "OK";

        // Calculate prior 24h runtime (in minutes, at 0.5 min per 30s row)
        var now = new Date();
        var cutoffTime24h = new Date(now.getTime() - (24 * 60 * 60 * 1000));
        var w1Mins = 0, p1Mins = 0, p2Mins = 0;

        for (var i = histData.length - 1; i >= 0; i--) {
          var rowDate = new Date(histData[i][0]);
          if (rowDate >= cutoffTime24h) {
            if (histData[i][5] == 1)  w1Mins += 0.5;
            if (histData[i][9] == 1)  p1Mins += 0.5;
            if (histData[i][10] == 1) p2Mins += 0.5;
          } else {
            break;
          }
        }

        var w1Hrs = (w1Mins / 60).toFixed(1);
        var smallBoosterHrs = (p1Mins / 60).toFixed(1);
        var bigBoosterHrs   = (p2Mins / 60).toFixed(1);

        snapshotLines = [
          "Latest Log: " + rowTime,
          "• 4,000 Gal Reservoir: " + tkSm + " gal (Hi: " + ltHi + " | Lo: " + ltLo + ")",
          "• 70,000 Gal Reservoir: " + tkLg + " gal (Hi: " + utHi + " | Lo: " + utLo + ")",
          "• Cl2: " + cl2 + " mg/L",
          "• Flow: " + flow + " GPM",
          "• Well 1 (24h): " + w1Hrs + "h",
          "• Boosters (24h): 7.5hp:" + smallBoosterHrs + "h | 5hp:" + bigBoosterHrs + "h",
          "• System Alarm: " + alm
        ];
      }

      var testText = "CVEWS System Messaging Daily Test Text. (Some analog values aren't being monitored yet)";
      if (snapshotLines.length > 0) {
        testText += "\n\n" + snapshotLines.join("\n");
      }
      testText += "\n\nDashboard: https://htijohnson.github.io/CVEWS/";

      dispatchSmsToRecipients(testText);
      return ContentService.createTextOutput(JSON.stringify({"status": "daily_test_sms_dispatched"}))
                           .setMimeType(ContentService.MimeType.JSON);
    }

    // Check for Alarm Acknowledgment SMS
    if (payloadData.action === "alarm_ack") {
      var techName = String(payloadData.name || "").trim();
      var enteredPin = String(payloadData.pin || "").trim();

      if (!validateRecipientCredentials(techName, enteredPin)) {
        return ContentService.createTextOutput(JSON.stringify({ "status": "error", "message": "Invalid Name or PIN" }))
                             .setMimeType(ContentService.MimeType.JSON);
      }

      var ackMessage = techName + " Acknowledged the Alarm and is investigating";
      dispatchSmsToRecipients(ackMessage);

      return ContentService.createTextOutput(JSON.stringify({ "status": "success", "message": ackMessage }))
                           .setMimeType(ContentService.MimeType.JSON);
    }

    // Standard Telemetry Logging to Current Monthly Sheet (yyyy-MM)
    var now = new Date();
    var sheet = getMonthlySheet(ss, now);
    
    var logRow = [
      now,
      payloadData.tk_sm,
      payloadData.tk_lg,
      payloadData.flow,
      payloadData.cl2,
      payloadData.w1 ? 1 : 0,
      payloadData.w3 ? 1 : 0,
      payloadData.w4 ? 1 : 0,
      payloadData.w5 ? 1 : 0,
      payloadData.p1 ? 1 : 0,
      payloadData.p2 ? 1 : 0,
      payloadData.alm ? 1 : 0,
      payloadData.device || "Lower Pumphouse",
      payloadData.ip || "",
      payloadData.version || "",
      payloadData.ut_hi ? 1 : 0,
      payloadData.ut_lo ? 1 : 0,
      payloadData.lt_hi ? 1 : 0,
      payloadData.lt_lo ? 1 : 0
    ];
    
    sheet.appendRow(logRow);
    
    try {
      scanForActiveAlarms(ss, sheet, logRow);
    } catch(err) {
      Logger.log("Alarm notifier error: " + err.toString());
    }
    
    return ContentService.createTextOutput(JSON.stringify({"status": "success", "sheet": sheet.getName(), "row": sheet.getLastRow()}))
                         .setMimeType(ContentService.MimeType.JSON);
                         
  } catch(error) {
    Logger.log("Critical Post Ingestion Failure: " + error.toString());
    return ContentService.createTextOutput(JSON.stringify({"status": "error", "message": error.toString()}))
                         .setMimeType(ContentService.MimeType.JSON);
  } finally {
    mLock.releaseLock();
  }
}

// 2. ACTIVE SYSTEM ALARM THREAD ENGINE
// ==========================================
function scanForActiveAlarms(ss, sheet, logRow) {
  ss = ss || SpreadsheetApp.getActiveSpreadsheet();
  sheet = sheet || getMonthlySheet(ss);

  var recipientSheet = ss.getSheetByName("Recipients");
  var notifyTargets = "";
  
  if (recipientSheet) {
    var lastRecipientRow = recipientSheet.getLastRow();
    if (lastRecipientRow >= 1) {
      var colToRead = 1;
      var firstCell = recipientSheet.getRange(1, 1).getValue().toString().toLowerCase();
      if (firstCell === "timestamp") {
        colToRead = 3; 
      }

      var startRow = (firstCell === "timestamp" || firstCell === "email" || firstCell === "recipient") ? 2 : 1;
      var numRows = lastRecipientRow - startRow + 1;

      if (numRows > 0) {
        var recipientValues = recipientSheet.getRange(startRow, colToRead, numRows, 1).getValues();
        
        var cleanList = recipientValues.map(function(row) {
          return row[0] ? row[0].toString().trim() : "";
        }).filter(function(target) {
          return target.length > 0 && target.indexOf("@") !== -1;
        });
        
        notifyTargets = cleanList.join(", ");
      }
    }
  }
  
  if (!notifyTargets) return;

  var timestamp, tk_sm, tk_lg, flow, alarmStatus;
  if (logRow) {
    timestamp   = logRow[0];
    tk_sm       = logRow[1];
    tk_lg       = logRow[2];
    flow        = logRow[3];
    alarmStatus = logRow[11];
  } else {
    var lastRow = sheet.getLastRow();
    if (lastRow < 2) return;
    var rowVals = sheet.getRange(lastRow, 1, 1, 12).getValues()[0];
    timestamp   = rowVals[0];
    tk_sm       = rowVals[1];
    tk_lg       = rowVals[2];
    flow        = rowVals[3];
    alarmStatus = rowVals[11];
  }

  // Track notification state in ScriptProperties so month rollovers never lose alarm state
  var isAlarmNotified = (scriptProps.getProperty("EMAIL_ALARM_NOTIFIED") === "true");
  var isTripped = (alarmStatus === 1 || alarmStatus === "1" || alarmStatus === "TRIPPED");
  var isClear   = (alarmStatus === 0 || alarmStatus === "0" || alarmStatus === "CLEAR");
  var formattedTime = Utilities.formatDate(new Date(timestamp), Session.getScriptTimeZone(), "MM/dd HH:mm:ss");

  if (isTripped && !isAlarmNotified) {
    var emailSubject = "⚠️ CVEWS ALARM!";
    var emailBody = "🚨 SYSTEM TRIPPED!\n" +
                    "Time: " + formattedTime + "\n" +
                    "• 4,000 Gal Reservoir: " + tk_sm + " Gal\n" +
                    "• 70,000 Gal Reservoir: " + tk_lg + " Gal\n" +
                    "• Flow: " + flow + " GPM\n" +
                    "Dash: https://htijohnson.github.io/CVEWS/"; 
                    
    try {
      MailApp.sendEmail(notifyTargets, emailSubject, emailBody);
      scriptProps.setProperty("EMAIL_ALARM_NOTIFIED", "true");
    } catch(e) {
      Logger.log("Email dispatch failed: " + e.toString());
    }
  }
  else if (isClear && isAlarmNotified) {
    var clearSubject = "✅ CVEWS RESOLVED";
    var clearBody = "💚 SYSTEM CLEAR / BACK TO NORMAL\n" +
                    "Cleared At: " + formattedTime + "\n\n" +
                    "• 4,000 Gal Reservoir: " + tk_sm + " Gal\n" +
                    "• 70,000 Gal Reservoir: " + tk_lg + " Gal\n" +
                    "• Flow: " + flow + " GPM\n" +
                    "System monitoring continuing normally.";
                    
    try {
      MailApp.sendEmail(notifyTargets, clearSubject, clearBody);
      scriptProps.deleteProperty("EMAIL_ALARM_NOTIFIED");
    } catch(e) {
      Logger.log("Email dispatch failed: " + e.toString());
    }
  }
}

// 3. CUSTOM VOLUMETRIC YIELD REPORT ENGINE
// ==========================================
function generateVolumetricYieldReport(ss, startIso, endIso, resolution) {
  var startDate = new Date(startIso);
  var endDate = new Date(endIso);
  if (isNaN(startDate.getTime()) || isNaN(endDate.getTime())) {
    return { error: "Invalid start or end date." };
  }

  var tz = Session.getScriptTimeZone();
  var formatPattern = "MM/dd/yyyy HH:00";
  if (resolution === "minute") {
    formatPattern = "MM/dd/yyyy HH:mm";
  } else if (resolution === "daily") {
    formatPattern = "MM/dd/yyyy";
  }

  var buckets = {};
  var bucketOrder = [];
  var grandTotalGallons = 0;

  // Iterate month-by-month across only the relevant monthly sheets
  var cursor = new Date(startDate.getFullYear(), startDate.getMonth(), 1);
  var endMonth = new Date(endDate.getFullYear(), endDate.getMonth(), 1);

  while (cursor <= endMonth) {
    var sheetName = Utilities.formatDate(cursor, tz, "yyyy-MM");
    var sheet = ss.getSheetByName(sheetName);

    if (sheet && sheet.getLastRow() >= 2) {
      var numRows = sheet.getLastRow() - 1;
      var values = sheet.getRange(2, 1, numRows, 4).getValues();

      for (var i = 0; i < numRows; i++) {
        var rowTime = new Date(values[i][0]);
        if (rowTime >= startDate && rowTime <= endDate) {
          var gpm = parseFloat(values[i][3]) || 0;
          var gallons = gpm * 0.5;
          grandTotalGallons += gallons;

          var bucketKey = Utilities.formatDate(rowTime, tz, formatPattern);
          if (!buckets[bucketKey]) {
            buckets[bucketKey] = {
              period: bucketKey,
              gallons: 0,
              gpmSum: 0,
              count: 0
            };
            bucketOrder.push(bucketKey);
          }
          buckets[bucketKey].gallons += gallons;
          buckets[bucketKey].gpmSum += gpm;
          buckets[bucketKey].count += 1;
        }
      }
    }
    cursor.setMonth(cursor.getMonth() + 1);
  }

  var rows = bucketOrder.map(function(key) {
    var b = buckets[key];
    return {
      period: b.period,
      avg_gpm: b.count > 0 ? (b.gpmSum / b.count) : 0,
      gallons: Math.round(b.gallons)
    };
  });

  return {
    resolution: resolution,
    total_gallons: Math.round(grandTotalGallons),
    rows: rows
  };
}

// 4. WEBPAGE API ENTRANCE (GET METHOD)
// ==========================================
function doGet(e) {
  // Handle Secret Control Dashboard Password Check (against SECRET_DASH_PASS Script Property)
  if (e && e.parameter && e.parameter.type === "verify_secret") {
    var submittedPass = String(e.parameter.pass || "").trim();
    var validPass = String(SECRET_DASH_PASS || "").trim();
    var isValid = (submittedPass.length > 0 && submittedPass === validPass);
    return ContentService.createTextOutput(JSON.stringify({ "valid": isValid }))
                         .setMimeType(ContentService.MimeType.JSON);
  }

  var ss = SpreadsheetApp.getActiveSpreadsheet();
  
  // Handle opt-in submissions via GET query params
  if (e && e.parameter && (e.parameter.type === "optin" || e.parameter.phone || e.parameter.name)) {
    var mLock = LockService.getScriptLock();
    try {
      mLock.waitLock(30000);
      var name = e.parameter.name || "";
      var phone = e.parameter.phone || "";
      var ip = e.parameter.ip || "Unknown";
      var consentText = e.parameter.consent_text || "";
      
      saveOptInRecord(ss, name, phone, ip, consentText);
      return ContentService.createTextOutput(JSON.stringify({ "status": "success", "message": "Opt-in saved successfully" }))
                           .setMimeType(ContentService.MimeType.JSON);
    } catch(err) {
      return ContentService.createTextOutput(JSON.stringify({ "status": "error", "message": err.toString() }))
                           .setMimeType(ContentService.MimeType.JSON);
    } finally {
      mLock.releaseLock();
    }
  }

  // Handle Custom Volumetric Yield Report Query
  if (e && e.parameter && e.parameter.type === "yield_report") {
    var reportResult = generateVolumetricYieldReport(
      ss,
      e.parameter.start,
      e.parameter.end,
      e.parameter.resolution || "hourly"
    );
    return ContentService.createTextOutput(JSON.stringify(reportResult))
                         .setMimeType(ContentService.MimeType.JSON);
  }
  
  var isRawRequested = (e && e.parameter && (e.parameter.type === "flow" || e.parameter.type === "raw" || e.parameter.format === "raw"));

  // Fetch up to 50,000 recent rows across current and previous monthly tabs in a single pass
  var dataRange = getRecentTelemetryRows(ss, 50000);

  if (dataRange.length === 0) {
    if (isRawRequested) {
      return ContentService.createTextOutput(JSON.stringify([])).setMimeType(ContentService.MimeType.JSON);
    }
    return ContentService.createTextOutput(JSON.stringify({ error: "No data available" })).setMimeType(ContentService.MimeType.JSON);
  }

  var nowMs = Date.now();
  var cutoff1h  = nowMs - (1 * 60 * 60 * 1000);
  var cutoff6h  = nowMs - (6 * 60 * 60 * 1000);
  var cutoff12h = nowMs - (12 * 60 * 60 * 1000);
  var cutoff24h = nowMs - (24 * 60 * 60 * 1000);
  var cutoff30d = nowMs - (30 * 24 * 60 * 60 * 1000);

  var pumped1h = 0, pumped6h = 0, pumped12h = 0, pumped24h = 0;
  var w1Mins = 0, w3Mins = 0, w4Mins = 0, w5Mins = 0, p1Mins = 0, p2Mins = 0;
  var smallTankValues = [];
  var largeTankValues = [];
  var flowValues      = [];
  var cl2Values       = [];
  var rawHistoricalLogs = [];

  var alarmTrippedCount = 0;
  var inAlarmState = false;

  var utHiTrips = 0, inUtHi = false;
  var utLoTrips = 0, inUtLo = false;
  var ltHiTrips = 0, inLtHi = false;
  var ltLoTrips = 0, inLtLo = false;

  for (var i = 0; i < dataRange.length; i++) {
    var rowTime = new Date(dataRange[i][0]);
    var rowMs = rowTime.getTime();
    
    if (rowMs >= cutoff30d) {
      var almStatus = dataRange[i][11];
      var isRowTripped = (almStatus === 1 || almStatus === "1" || almStatus === true || almStatus === "TRIPPED");
      if (isRowTripped) {
        if (!inAlarmState) { alarmTrippedCount++; inAlarmState = true; }
      } else {
        inAlarmState = false;
      }

      if (dataRange[i][15] === 1 || dataRange[i][15] === "1") {
        if (!inUtHi) { utHiTrips++; inUtHi = true; }
      } else { inUtHi = false; }

      if (dataRange[i][16] === 1 || dataRange[i][16] === "1") {
        if (!inUtLo) { utLoTrips++; inUtLo = true; }
      } else { inUtLo = false; }

      if (dataRange[i][17] === 1 || dataRange[i][17] === "1") {
        if (!inLtHi) { ltHiTrips++; inLtHi = true; }
      } else { inLtHi = false; }

      if (dataRange[i][18] === 1 || dataRange[i][18] === "1") {
        if (!inLtLo) { ltLoTrips++; inLtLo = true; }
      } else { inLtLo = false; }
    }

    if (rowMs >= cutoff24h) {
      var gpmFlow = parseFloat(dataRange[i][3]) || 0;
      var m3hFlow = (gpmFlow * 60.0) / M3_TO_GALLONS;
      var rowGallons = gpmFlow * 0.5;

      rawHistoricalLogs.push({
        timestamp: rowTime.toISOString(),
        flow_rate: m3hFlow
      });

      pumped24h += rowGallons;
      if (rowMs >= cutoff12h) pumped12h += rowGallons;
      if (rowMs >= cutoff6h)  pumped6h  += rowGallons;
      if (rowMs >= cutoff1h)  pumped1h  += rowGallons;

      if (dataRange[i][5] === 1)  w1Mins += 0.5;
      if (dataRange[i][6] === 1)  w3Mins += 0.5;
      if (dataRange[i][7] === 1)  w4Mins += 0.5;
      if (dataRange[i][8] === 1)  w5Mins += 0.5;
      if (dataRange[i][9] === 1)  p1Mins += 0.5;
      if (dataRange[i][10] === 1) p2Mins += 0.5;

      var smVal   = Number(dataRange[i][1]);
      var lgVal   = Number(dataRange[i][2]);
      var flVal   = Number(dataRange[i][3]);
      var cl2Val  = Number(dataRange[i][4]);

      if (!isNaN(smVal) && smVal > 0)   smallTankValues.push(smVal);
      if (!isNaN(lgVal) && lgVal > 0)   largeTankValues.push(lgVal);
      if (!isNaN(flVal) && flVal >= 0)  flowValues.push(flVal);
      if (!isNaN(cl2Val) && cl2Val >= 0) cl2Values.push(cl2Val);
    }
  }

  if (isRawRequested) {
    return ContentService.createTextOutput(JSON.stringify(rawHistoricalLogs))
                         .setMimeType(ContentService.MimeType.JSON);
  }

  var latestRow    = dataRange[dataRange.length - 1];
  var currentSmall = latestRow[1];
  var currentLarge = latestRow[2];
  var currentFlow  = latestRow[3];
  var currentCl2   = latestRow[4];
  
  var smMin   = smallTankValues.length ? Math.min.apply(null, smallTankValues) : currentSmall;
  var smMax   = smallTankValues.length ? Math.max.apply(null, smallTankValues) : currentSmall;
  var lgMin   = largeTankValues.length ? Math.min.apply(null, largeTankValues) : currentLarge;
  var lgMax   = largeTankValues.length ? Math.max.apply(null, largeTankValues) : currentLarge;
  var flowMin = flowValues.length      ? Math.min.apply(null, flowValues)      : currentFlow;
  var flowMax = flowValues.length      ? Math.max.apply(null, flowValues)      : currentFlow;
  var cl2Min  = cl2Values.length       ? Math.min.apply(null, cl2Values)       : currentCl2;
  var cl2Max  = cl2Values.length       ? Math.max.apply(null, cl2Values)       : currentCl2;
  
  var data = {
    "timestamp": Utilities.formatDate(new Date(latestRow[0]), Session.getScriptTimeZone(), "MM/dd HH:mm:ss"),
    "device":    latestRow[12] ? String(latestRow[12]) : "Lower Pumphouse",
    "ip":        latestRow[13] ? String(latestRow[13]) : "",
    "version":   latestRow[14] ? String(latestRow[14]) : "",
    "tk_sm":     currentSmall,
    "tk_lg":     currentLarge,
    "flow":      currentFlow,
    "cl2":       currentCl2,
    "w1":        latestRow[5],
    "w3":        latestRow[6],
    "w4":        latestRow[7],
    "w5":        latestRow[8],
    "p1":        latestRow[9],
    "p2":        latestRow[10],
    "alm":       latestRow[11],
    "ut_hi":     latestRow[15] ? 1 : 0,
    "ut_lo":     latestRow[16] ? 1 : 0,
    "lt_hi":     latestRow[17] ? 1 : 0,
    "lt_lo":     latestRow[18] ? 1 : 0,
    "pumped_1h":  Math.round(pumped1h),
    "pumped_6h":  Math.round(pumped6h),
    "pumped_12h": Math.round(pumped12h),
    "pumped_24h": Math.round(pumped24h),
    "w1_24h": (w1Mins / 60).toFixed(1),
    "w3_24h": (w3Mins / 60).toFixed(1),
    "w4_24h": (w4Mins / 60).toFixed(1),
    "w5_24h": (w5Mins / 60).toFixed(1),
    "p1_24h": (p1Mins / 60).toFixed(1),
    "p2_24h": (p2Mins / 60).toFixed(1),
    "sm_min":   Math.round(smMin),
    "sm_max":   Math.round(smMax),
    "lg_min":   Math.round(lgMin),
    "lg_max":   Math.round(lgMax),
    "flow_min": Number(flowMin || 0).toFixed(1),
    "flow_max": Number(flowMax || 0).toFixed(1),
    "cl2_min":  Number(cl2Min || 0).toFixed(2),
    "cl2_max":  Number(cl2Max || 0).toFixed(2),
    "alarm_30d_count": alarmTrippedCount,
    "ut_hi_30d": utHiTrips,
    "ut_lo_30d": utLoTrips,
    "lt_hi_30d": ltHiTrips,
    "lt_lo_30d": ltLoTrips
  };
  
  return ContentService.createTextOutput(JSON.stringify(data)).setMimeType(ContentService.MimeType.JSON);
}

// 5A. COMPILER ENGINE (Run at 11:59 PM PST)
// ==========================================
function compileDailyReportData() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var dataRange = getRecentTelemetryRows(ss, 3000);
  if (dataRange.length === 0) return;

  var now = new Date();
  var cutoffTime = new Date(now.getTime() - (24 * 60 * 60 * 1000));
  
  var totalGallons = 0;
  var minutesW1 = 0, minutesW3 = 0, minutesW4 = 0, minutesW5 = 0;
  var minutesP1 = 0, minutesP2 = 0;
  var actualRowsAnalyzed = 0;

  for (var i = dataRange.length - 1; i >= 0; i--) {
    var rowTime = new Date(dataRange[i][0]);
    if (rowTime >= cutoffTime) {
      actualRowsAnalyzed++;
      totalGallons += ((parseFloat(dataRange[i][3]) || 0) * 0.5);
      if (dataRange[i][5] === 1)  minutesW1 += 0.5;
      if (dataRange[i][6] === 1)  minutesW3 += 0.5;
      if (dataRange[i][7] === 1)  minutesW4 += 0.5;
      if (dataRange[i][8] === 1)  minutesW5 += 0.5;
      if (dataRange[i][9] === 1)  minutesP1 += 0.5;
      if (dataRange[i][10] === 1) minutesP2 += 0.5;
    } else {
      break;
    }
  }

  var cacheSheet = ss.getSheetByName("ReportCache");
  if (!cacheSheet) {
    cacheSheet = ss.insertSheet("ReportCache");
  }
  cacheSheet.clear();
  
  var dateString = Utilities.formatDate(now, "PST", "MM/dd/yyyy");
  
  cacheSheet.appendRow([
    dateString, totalGallons, actualRowsAnalyzed,
    Math.round(minutesW1), Math.round(minutesW3), Math.round(minutesW4), Math.round(minutesW5),
    Math.round(minutesP1), Math.round(minutesP2)
  ]);
}

// Run this ONCE manually from the Apps Script editor to install the monthly trigger
function setupMonthlyArchiveTrigger() {
  var triggers = ScriptApp.getProjectTriggers();
  for (var i = 0; i < triggers.length; i++) {
    if (triggers[i].getHandlerFunction() === "archivePreviousMonthSheet") {
      ScriptApp.deleteTrigger(triggers[i]);
    }
  }

  // Runs on the 1st of every month at 2:00 AM
  ScriptApp.newTrigger("archivePreviousMonthSheet")
    .timeBased()
    .onMonthDay(1)
    .atHour(2)
    .create();

  Logger.log("Monthly archival trigger installed: archivePreviousMonthSheet will run on day 1 at 2:00 AM.");
}

// Exports the previous month's tab (yyyy-MM) to a standalone archive file in Drive,
// then removes that tab from the active workbook to prevent cell-limit bloat.
function archivePreviousMonthSheet() {
  var mLock = LockService.getScriptLock();
  try {
    mLock.waitLock(30000);

    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var tz = Session.getScriptTimeZone();
    var now = new Date();

    // Ensure current month's sheet exists before deleting any old sheet
    getMonthlySheet(ss, now);

    // Target the previous month (e.g., if today is Dec 1, targets "2026-11")
    var prevMonthDate = new Date(now.getFullYear(), now.getMonth() - 1, 15);
    var prevSheetName = Utilities.formatDate(prevMonthDate, tz, "yyyy-MM");
    var prevSheet = ss.getSheetByName(prevSheetName);

    if (!prevSheet) {
      Logger.log("No sheet found for previous month (" + prevSheetName + "). Nothing to archive.");
      return;
    }

    // Locate or create the archive folder in Google Drive
    var folders = DriveApp.getFoldersByName(ARCHIVE_FOLDER_NAME);
    var archiveFolder = folders.hasNext() ? folders.next() : DriveApp.createFolder(ARCHIVE_FOLDER_NAME);

    // Create standalone archive spreadsheet and copy the month's sheet into it
    var archiveFileName = "CVEWS_Telemetry_" + prevSheetName;
    var archiveSs = SpreadsheetApp.create(archiveFileName);
    var copiedSheet = prevSheet.copyTo(archiveSs);
    copiedSheet.setName(prevSheetName);

    // Remove the default blank "Sheet1" from the new archive spreadsheet
    var defaultSheet = archiveSs.getSheetByName("Sheet1");
    if (defaultSheet && archiveSs.getSheets().length > 1) {
      archiveSs.deleteSheet(defaultSheet);
    }

    // Move archive spreadsheet into the dedicated Drive folder
    var archiveFile = DriveApp.getFileById(archiveSs.getId());
    archiveFile.moveTo(archiveFolder);

    // Delete the archived sheet from the live logger workbook (only if > 1 sheet remains)
    if (ss.getSheets().length > 1) {
      ss.deleteSheet(prevSheet);
    }

    Logger.log("Successfully archived and removed sheet: " + prevSheetName + " -> " + archiveFileName);
  } catch (err) {
    Logger.log("Monthly archival error: " + err.toString());
  } finally {
    mLock.releaseLock();
  }
}
