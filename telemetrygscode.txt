// TWILIO CREDENTIALS CONFIGURATION
// Stored securely in Project Settings > Script Properties
// ==========================================
var scriptProps = PropertiesService.getScriptProperties();
var TWILIO_ACCOUNT_SID = scriptProps.getProperty("TWILIO_ACCOUNT_SID");
var TWILIO_AUTH_TOKEN  = scriptProps.getProperty("TWILIO_AUTH_TOKEN");
var TWILIO_FROM_NUMBER = scriptProps.getProperty("TWILIO_FROM_NUMBER");
var ALARM_ACK_PIN      = scriptProps.getProperty("ALARM_ACK_PIN") || "3141";

var M3_TO_GALLONS = 264.172;

// Helper to format any standard US phone string into E.164 (+1XXXXXXXXXX)
function formatToE164(phone) {
  if (!phone) return null;
  var digits = String(phone).replace(/\D/g, "");
  if (digits.length === 10) {
    return "+1" + digits;
  } else if (digits.length === 11 && digits.charAt(0) === "1") {
    return "+" + digits;
  } else if (digits.length > 10) {
    return "+" + digits;
  }
  return null;
}

// Validates that PIN matches the last 4 of Column 3 (Phone) AND
// entered name matches (or is a part of) Column 2 (Full Name) on that same row
function validateRecipientCredentials(enteredName, enteredPin) {
  var cleanPin = String(enteredPin || "").trim();
  var cleanName = String(enteredName || "").trim().toLowerCase();

  if (!cleanPin || !cleanName) return false;

  // Master/Admin PIN override check
  if (cleanPin === String(ALARM_ACK_PIN).trim()) {
    return true;
  }

  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var recipientSheet = ss.getSheetByName("Recipients");
  if (!recipientSheet) return false;

  var lastRow = recipientSheet.getLastRow();
  if (lastRow < 2) return false;

  // Read Column 2 (Full Name) and Column 3 (Phone / Target)
  var data = recipientSheet.getRange(2, 2, lastRow - 1, 2).getValues();

  for (var i = 0; i < data.length; i++) {
    var sheetName = String(data[i][0] || "").trim().toLowerCase();
    var rawPhone = String(data[i][1] || "").replace(/\D/g, "");

    if (rawPhone.length >= 4) {
      var lastFour = rawPhone.slice(-4);

      if (cleanPin === lastFour) {
        if (sheetName.indexOf(cleanName) !== -1 || cleanName.indexOf(sheetName) !== -1) {
          return true;
        }
      }
    }
  }

  return false;
}

function sendTwilioSms(toPhone, bodyText) {
  var formattedTo = formatToE164(toPhone);
  if (!formattedTo) {
    Logger.log("Skipping SMS dispatch: Invalid phone number format: " + toPhone);
    return;
  }

  var url = "https://api.twilio.com/2010-04-01/Accounts/" + TWILIO_ACCOUNT_SID + "/Messages.json";
  var options = {
    "method": "post",
    "headers": {
      "Authorization": "Basic " + Utilities.base64Encode(TWILIO_ACCOUNT_SID + ":" + TWILIO_AUTH_TOKEN)
    },
    "payload": {
      "To": formattedTo,
      "From": TWILIO_FROM_NUMBER,
      "Body": bodyText
    },
    "muteHttpExceptions": true
  };
  try {
    var response = UrlFetchApp.fetch(url, options);
    var code = response.getResponseCode();
    var text = response.getContentText();
    Logger.log("Twilio HTTP " + code + " to " + formattedTo + ": " + text);
  } catch(e) {
    Logger.log("Error dispatching SMS to " + formattedTo + ": " + e.toString());
  }
}

function dispatchSmsToRecipients(messageBody) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var recipientSheet = ss.getSheetByName("Recipients");
  if (!recipientSheet) {
    Logger.log("Recipients sheet missing. SMS aborted.");
    return;
  }
  
  var lastRow = recipientSheet.getLastRow();
  if (lastRow < 2) return;
  
  var data = recipientSheet.getRange(2, 1, lastRow - 1, 5).getValues();
  
  for (var i = 0; i < data.length; i++) {
    var phone = String(data[i][2]).trim();
    if (phone) {
      sendTwilioSms(phone, messageBody);
    }
  }
}

// ==========================================
// OPT-IN RECORD WRITER (USED BY GET & POST)
// ==========================================
function saveOptInRecord(ss, name, phone, ip, consentText) {
  var recipientSheet = ss.getSheetByName("Recipients");
  if (!recipientSheet) {
    recipientSheet = ss.insertSheet("Recipients");
  }
  
  if (recipientSheet.getLastRow() === 0) {
    recipientSheet.appendRow(["Timestamp", "Full Name", "Phone / Target", "IP Address", "Consent Disclosure"]);
    recipientSheet.getRange("A1:E1").setFontWeight("bold");
  }

  var formattedPhone = formatToE164(phone) || phone;
  recipientSheet.appendRow([new Date(), name, formattedPhone, ip, consentText]);
  SpreadsheetApp.flush();
  Logger.log("Successfully recorded opt-in for: " + name + " (" + formattedPhone + ")");
}

// ==========================================
// 1. DATA LOGGER & FORM INGESTION ENTRY POINT
// ==========================================
function doPost(e) {
  var mLock = LockService.getScriptLock();
  try {
    mLock.waitLock(30000);
    
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var rawContents = (e && e.postData && e.postData.contents) ? e.postData.contents : "";
    Logger.log("Incoming doPost rawContents: " + rawContents);

    var parsedJson = null;
    if (rawContents) {
      try {
        parsedJson = JSON.parse(rawContents);
      } catch (parseIgnore) {}
    }

    // ----------------------------------------------------
    // CHECK 1: Opt-in Form Submission
    // ----------------------------------------------------
    var isOptIn = false;
    var name = "", phone = "", ip = "Unknown", consentText = "";

    if (parsedJson && (parsedJson.type === "optin" || parsedJson.phone || parsedJson.consent_text)) {
      isOptIn = true;
      name = parsedJson.name || "";
      phone = parsedJson.phone || "";
      ip = parsedJson.ip || "Unknown";
      consentText = parsedJson.consent_text || "";
    } else if (e && e.parameter && (e.parameter.type === "optin" || e.parameter.phone || e.parameter.name)) {
      isOptIn = true;
      name = e.parameter.name || "";
      phone = e.parameter.phone || "";
      ip = e.parameter.ip || "Unknown";
      consentText = e.parameter.consent_text || "";
    } else if (rawContents && (rawContents.indexOf("type=optin") !== -1 || rawContents.indexOf("phone=") !== -1)) {
      isOptIn = true;
      var params = {};
      rawContents.split("&").forEach(function(part) {
        var item = part.split("=");
        if (item.length === 2) {
          params[decodeURIComponent(item[0])] = decodeURIComponent(item[1].replace(/\+/g, " "));
        }
      });
      name = params.name || "";
      phone = params.phone || "";
      ip = params.ip || "Unknown";
      consentText = params.consent_text || "";
    }

    if (isOptIn) {
      saveOptInRecord(ss, name, phone, ip, consentText);
      return ContentService.createTextOutput(JSON.stringify({ "status": "success", "message": "Opt-in saved successfully" }))
                           .setMimeType(ContentService.MimeType.JSON);
    }

    // ----------------------------------------------------
    // CHECK 2: ESP32 JSON Payload Ingestion
    // ----------------------------------------------------
    var payloadData = parsedJson;
    if (!payloadData) {
      Logger.log("Invalid or unrecognized data: " + rawContents);
      return ContentService.createTextOutput(JSON.stringify({"status": "error", "message": "Invalid JSON or unrecognized form data"}))
                           .setMimeType(ContentService.MimeType.JSON);
    }
    
    // Check for Alarm SMS Trigger (> 30s)
    if (payloadData.action === "alarm_sms") {
      var alertText = payloadData.message || "🚨 CVEWS CRITICAL ALERT: System Alarm Light active > 30s! Check https://htijohnson.github.io/CVEWS/";
      dispatchSmsToRecipients(alertText);
      return ContentService.createTextOutput(JSON.stringify({"status": "alarm_sms_dispatched"}))
                           .setMimeType(ContentService.MimeType.JSON);
    }

    // Check for Alarm Resolved SMS Trigger
    if (payloadData.action === "alarm_clear") {
      var clearText = payloadData.message || "✅ CVEWS RESOLVED: The System Alarm has cleared and returned to normal status.";
      dispatchSmsToRecipients(clearText);
      return ContentService.createTextOutput(JSON.stringify({"status": "alarm_clear_dispatched"}))
                           .setMimeType(ContentService.MimeType.JSON);
    }

    // Check for Daily Noon Test SMS
    if (payloadData.action === "daily_test_sms") {
      var testText = payloadData.message || "CVEWS DAILY SYSTEM TEST: System operational. Dashboard: https://htijohnson.github.io/CVEWS/";
      dispatchSmsToRecipients(testText);
      return ContentService.createTextOutput(JSON.stringify({"status": "daily_test_sms_dispatched"}))
                           .setMimeType(ContentService.MimeType.JSON);
    }

    // Check for Alarm Acknowledgment SMS
    if (payloadData.action === "alarm_ack") {
      var techName = String(payloadData.name || "").trim();
      var enteredPin = String(payloadData.pin || "").trim();

      if (!validateRecipientCredentials(techName, enteredPin)) {
        return ContentService.createTextOutput(JSON.stringify({ "status": "error", "message": "Invalid Name or PIN" }))
                             .setMimeType(ContentService.MimeType.JSON);
      }

      var ackMessage = techName + " Acknowledge the Alarm and is investingating";
      dispatchSmsToRecipients(ackMessage);

      return ContentService.createTextOutput(JSON.stringify({ "status": "success", "message": ackMessage }))
                           .setMimeType(ContentService.MimeType.JSON);
    }

    // Standard Telemetry Logging to Sheet1
    var sheet = ss.getSheetByName("Sheet1");
    if (!sheet) {
      sheet = ss.insertSheet("Sheet1");
    }
    
    var logRow = [
      new Date(),
      payloadData.tk_sm,
      payloadData.tk_lg,
      payloadData.flow,
      payloadData.cl2,
      payloadData.w1 ? 1 : 0,
      payloadData.w3 ? 1 : 0,
      payloadData.w4 ? 1 : 0,
      payloadData.w5 ? 1 : 0,
      payloadData.p1 ? 1 : 0,
      payloadData.p2 ? 1 : 0,
      payloadData.alm ? 1 : 0
    ];
    
    sheet.appendRow(logRow);
    
    try {
      scanForActiveAlarms(payloadData);
    } catch(err) {
      Logger.log("Alarm notifier error: " + err.toString());
    }
    
    return ContentService.createTextOutput(JSON.stringify({"status": "success", "row": sheet.getLastRow()}))
                         .setMimeType(ContentService.MimeType.JSON);
                         
  } catch(error) {
    Logger.log("Critical Post Ingestion Failure: " + error.toString());
    return ContentService.createTextOutput(JSON.stringify({"status": "error", "message": error.toString()}))
                         .setMimeType(ContentService.MimeType.JSON);
  } finally {
    mLock.releaseLock();
  }
}

// 2. ACTIVE SYSTEM ALARM THREAD ENGINE
// ==========================================
function scanForActiveAlarms(payloadData) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName("Sheet1");
  var lastRow = sheet.getLastRow();
  
  if (lastRow < 2) return; 
  
  var recipientSheet = ss.getSheetByName("Recipients");
  var notifyTargets = "";
  
  if (recipientSheet) {
    var lastRecipientRow = recipientSheet.getLastRow();
    if (lastRecipientRow >= 1) {
      var colToRead = 1;
      var firstCell = recipientSheet.getRange(1, 1).getValue().toString().toLowerCase();
      if (firstCell === "timestamp") {
        colToRead = 3; 
      }

      var startRow = (firstCell === "timestamp" || firstCell === "email" || firstCell === "recipient") ? 2 : 1;
      var numRows = lastRecipientRow - startRow + 1;

      if (numRows > 0) {
        var recipientValues = recipientSheet.getRange(startRow, colToRead, numRows, 1).getValues();
        
        var cleanList = recipientValues.map(function(row) {
          return row[0] ? row[0].toString().trim() : "";
        }).filter(function(target) {
          return target.length > 0 && target.indexOf("@") !== -1;
        });
        
        notifyTargets = cleanList.join(", ");
      }
    }
  }
  
  if (!notifyTargets) return;

  var lastAlarmCell = sheet.getRange(lastRow, 12); 
  var alarmStatus = lastAlarmCell.getValue();
  var currentNote = lastAlarmCell.getNote();
  
  var previousAlarmCell = sheet.getRange(lastRow - 1, 12);
  var previousNote = previousAlarmCell.getNote();
  
  var timestamp = sheet.getRange(lastRow, 1).getValue();  
  var tk_sm     = sheet.getRange(lastRow, 2).getValue();  
  var tk_lg     = sheet.getRange(lastRow, 3).getValue();  
  var flow      = sheet.getRange(lastRow, 4).getValue();  
  
  var formattedTime = Utilities.formatDate(new Date(timestamp), Session.getScriptTimeZone(), "MM/dd HH:mm:ss");

  if ((alarmStatus === 1 || alarmStatus === "1" || alarmStatus === "TRIPPED") && currentNote !== "NOTIFIED") {
    var emailSubject = "⚠️ CVEWS ALARM!";
    var emailBody = "🚨 SYSTEM TRIPPED!\n" +
                    "Time: " + formattedTime + "\n" +
                    "• Sm Tank: " + tk_sm + " Gal\n" +
                    "• Lg Tank: " + tk_lg + " Gal\n" +
                    "• Flow: " + flow + " GPM\n" +
                    "Dash: https://htijohnson.github.io/CVEWS/"; 
                    
    try {
      MailApp.sendEmail(notifyTargets, emailSubject, emailBody);
      lastAlarmCell.setNote("NOTIFIED");
    } catch(e) {
      Logger.log("Email dispatch failed: " + e.toString());
    }
  }
  else if ((alarmStatus === 0 || alarmStatus === "0" || alarmStatus === "CLEAR") && previousNote === "NOTIFIED") {
    var clearSubject = "✅ CVEWS RESOLVED";
    var clearBody = "💚 SYSTEM CLEAR / BACK TO NORMAL\n" +
                    "Cleared At: " + formattedTime + "\n\n" +
                    "• Sm Tank: " + tk_sm + " Gal\n" +
                    "• Lg Tank: " + tk_lg + " Gal\n" +
                    "• Flow: " + flow + " GPM\n" +
                    "System monitoring continuing normally.";
                    
    try {
      MailApp.sendEmail(notifyTargets, clearSubject, clearBody);
      previousAlarmCell.clearNote();
    } catch(e) {
      Logger.log("Email dispatch failed: " + e.toString());
    }
  }
}

// 3. TOTALIZER INTEGRATION CODE ENGINE
// ==========================================
function getGallonsPumpedOverHours(hours) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName("Sheet1");
  var lastRow = sheet.getLastRow();
  if (lastRow < 2) return 0;

  var now = new Date();
  var cutoffTime = new Date(now.getTime() - (hours * 60 * 60 * 1000));
  
  // ESP32 posts every 30 seconds (~120 rows/hour)
  var maxRowsToRead = Math.min(lastRow - 1, hours * 130); 
  var startRow = lastRow - maxRowsToRead + 1;
  
  var timestamps = sheet.getRange(startRow, 1, maxRowsToRead, 1).getValues();
  var flowRates = sheet.getRange(startRow, 4, maxRowsToRead, 1).getValues();
  
  var totalGallons = 0;

  for (var i = timestamps.length - 1; i >= 0; i--) {
    var rowTime = new Date(timestamps[i][0]);
    if (rowTime >= cutoffTime) {
      var gpm = parseFloat(flowRates[i][0]) || 0;
      // 30 seconds = 0.5 minutes: gallons = GPM * 0.5
      totalGallons += (gpm * 0.5); 
    } else {
      break; 
    }
  }
  return Math.round(totalGallons);
}

// 4. WEBPAGE API ENTRANCE (GET METHOD)
// ==========================================
function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  
  // Handle opt-in submissions via GET query params
  if (e && e.parameter && (e.parameter.type === "optin" || e.parameter.phone || e.parameter.name)) {
    var mLock = LockService.getScriptLock();
    try {
      mLock.waitLock(30000);
      var name = e.parameter.name || "";
      var phone = e.parameter.phone || "";
      var ip = e.parameter.ip || "Unknown";
      var consentText = e.parameter.consent_text || "";
      
      saveOptInRecord(ss, name, phone, ip, consentText);
      return ContentService.createTextOutput(JSON.stringify({ "status": "success", "message": "Opt-in saved successfully" }))
                           .setMimeType(ContentService.MimeType.JSON);
    } catch(err) {
      return ContentService.createTextOutput(JSON.stringify({ "status": "error", "message": err.toString() }))
                           .setMimeType(ContentService.MimeType.JSON);
    } finally {
      mLock.releaseLock();
    }
  }

  var sheet = ss.getSheetByName("Sheet1");
  var lastRow = sheet.getLastRow();
  
  var isRawRequested = (e && e.parameter && (e.parameter.type === "flow" || e.parameter.type === "raw" || e.parameter.format === "raw"));

  if (lastRow < 2) {
    if (isRawRequested) {
      return ContentService.createTextOutput(JSON.stringify([])).setMimeType(ContentService.MimeType.JSON);
    }
    return ContentService.createTextOutput(JSON.stringify({ error: "No data available" })).setMimeType(ContentService.MimeType.JSON);
  }

  var now = new Date();
  var cutoffTime24h = new Date(now.getTime() - (24 * 60 * 60 * 1000));
  var cutoffTime30d = new Date(now.getTime() - (30 * 24 * 60 * 60 * 1000));
  
  var maxRowsToRead = Math.min(lastRow - 1, 50000); 
  var startRow = lastRow - maxRowsToRead + 1;
  var dataRange = sheet.getRange(startRow, 1, maxRowsToRead, 12).getValues();

  var w1Mins = 0, w3Mins = 0, w4Mins = 0, w5Mins = 0, p1Mins = 0, p2Mins = 0;
  var smallTankValues = [];
  var largeTankValues = [];
  var rawHistoricalLogs = [];

  var alarmTrippedCount = 0;
  var inAlarmState = false;

  for (var i = 0; i < dataRange.length; i++) {
    var rawDate = dataRange[i][0];
    var rowTime = new Date(rawDate);
    
    // Collect flow samples from the last 24 hours for index.html calculateIntegratedVolume()
    if (rowTime >= cutoffTime24h) {
      var gpmFlow = parseFloat(dataRange[i][3]) || 0;
      // Convert GPM to m3/h so client calculation ((rate * 264.172) / 60) results in exact gallons
      var m3hFlow = (gpmFlow * 60.0) / M3_TO_GALLONS;

      rawHistoricalLogs.push({
        timestamp: rowTime.toISOString(),
        flow_rate: m3hFlow
      });
    }

    if (rowTime >= cutoffTime30d) {
      var almStatus = dataRange[i][11];
      var isRowTripped = (almStatus === 1 || almStatus === "1" || almStatus === true || almStatus === "TRIPPED");
      
      if (isRowTripped) {
        if (!inAlarmState) {
          alarmTrippedCount++;
          inAlarmState = true;
        }
      } else {
        inAlarmState = false;
      }
    }

    if (rowTime >= cutoffTime24h) {
      // 30-second intervals = 0.5 mins per tick
      if (dataRange[i][5] === 1)  w1Mins += 0.5;
      if (dataRange[i][6] === 1)  w3Mins += 0.5;
      if (dataRange[i][7] === 1)  w4Mins += 0.5;
      if (dataRange[i][8] === 1)  w5Mins += 0.5;
      if (dataRange[i][9] === 1)  p1Mins += 0.5;
      if (dataRange[i][10] === 1) p2Mins += 0.5;

      var smVal = Number(dataRange[i][1]);
      var lgVal = Number(dataRange[i][2]);
      if (!isNaN(smVal) && smVal > 0) smallTankValues.push(smVal);
      if (!isNaN(lgVal) && lgVal > 0) largeTankValues.push(lgVal);
    }
  }

  // If index.html calls TUF_GOOGLE_APP_URL with ?type=flow
  if (isRawRequested) {
    return ContentService.createTextOutput(JSON.stringify(rawHistoricalLogs))
                         .setMimeType(ContentService.MimeType.JSON);
  }

  var currentSmall = sheet.getRange(lastRow, 2).getValue();
  var currentLarge = sheet.getRange(lastRow, 3).getValue();
  
  var smMin = smallTankValues.length ? Math.min.apply(null, smallTankValues) : currentSmall;
  var smMax = smallTankValues.length ? Math.max.apply(null, smallTankValues) : currentSmall;
  var lgMin = largeTankValues.length ? Math.min.apply(null, largeTankValues) : currentLarge;
  var lgMax = largeTankValues.length ? Math.max.apply(null, largeTankValues) : currentLarge;
  
  var data = {
    "timestamp": Utilities.formatDate(new Date(sheet.getRange(lastRow, 1).getValue()), Session.getScriptTimeZone(), "MM/dd HH:mm:ss"),
    "tk_sm":     currentSmall,
    "tk_lg":     currentLarge,
    "flow":      sheet.getRange(lastRow, 4).getValue(),
    "cl2":       sheet.getRange(lastRow, 5).getValue(),
    "pumped_1h":  getGallonsPumpedOverHours(1),
    "pumped_6h":  getGallonsPumpedOverHours(6),
    "pumped_12h": getGallonsPumpedOverHours(12),
    "pumped_24h": getGallonsPumpedOverHours(24),
    "w1_24h": (w1Mins / 60).toFixed(1),
    "w3_24h": (w3Mins / 60).toFixed(1),
    "w4_24h": (w4Mins / 60).toFixed(1),
    "w5_24h": (w5Mins / 60).toFixed(1),
    "p1_24h": (p1Mins / 60).toFixed(1),
    "p2_24h": (p2Mins / 60).toFixed(1),
    "sm_min": Math.round(smMin),
    "sm_max": Math.round(smMax),
    "lg_min": Math.round(lgMin),
    "lg_max": Math.round(lgMax),
    "alarm_30d_count": alarmTrippedCount
  };
  
  return ContentService.createTextOutput(JSON.stringify(data)).setMimeType(ContentService.MimeType.JSON);
}

// 5A. COMPILER ENGINE (Run at 11:59 PM PST)
// ==========================================
function compileDailyReportData() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName("Sheet1");
  var lastRow = sheet.getLastRow();
  if (lastRow < 2) return;

  var now = new Date();
  var cutoffTime = new Date(now.getTime() - (24 * 60 * 60 * 1000));
  
  var maxRowsToRead = Math.min(lastRow - 1, 3000); 
  var startRow = lastRow - maxRowsToRead + 1;
  var dataRange = sheet.getRange(startRow, 1, maxRowsToRead, 12).getValues();
  
  var totalGallons = 0;
  var minutesW1 = 0, minutesW3 = 0, minutesW4 = 0, minutesW5 = 0;
  var minutesP1 = 0, minutesP2 = 0;
  var actualRowsAnalyzed = 0;

  for (var i = dataRange.length - 1; i >= 0; i--) {
    var rowTime = new Date(dataRange[i][0]);
    if (rowTime >= cutoffTime) {
      actualRowsAnalyzed++;
      totalGallons += ((parseFloat(dataRange[i][3]) || 0) * 0.5);
      if (dataRange[i][5] === 1)  minutesW1 += 0.5;
      if (dataRange[i][6] === 1)  minutesW3 += 0.5;
      if (dataRange[i][7] === 1)  minutesW4 += 0.5;
      if (dataRange[i][8] === 1)  minutesW5 += 0.5;
      if (dataRange[i][9] === 1)  minutesP1 += 0.5;
      if (dataRange[i][10] === 1) minutesP2 += 0.5;
    } else {
      break;
    }
  }

  var cacheSheet = ss.getSheetByName("ReportCache");
  if (!cacheSheet) {
    cacheSheet = ss.insertSheet("ReportCache");
  }
  cacheSheet.clear();
  
  var dateString = Utilities.formatDate(now, "PST", "MM/dd/yyyy");
  
  cacheSheet.appendRow([
    dateString, totalGallons, actualRowsAnalyzed,
    Math.round(minutesW1), Math.round(minutesW3), Math.round(minutesW4), Math.round(minutesW5),
    Math.round(minutesP1), Math.round(minutesP2)
  ]);
}

// 5B. DISPATCHER ENGINE (Run at 1:00 AM PST)
// ==========================================
function dispatchDailyReportEmail() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var cacheSheet = ss.getSheetByName("ReportCache");
  
  if (!cacheSheet || cacheSheet.getLastRow() < 1) return;
  
  var cachedData = cacheSheet.getRange(1, 1, 1, 9).getValues()[0];
  var dateString      = cachedData[0];
  var totalGallons    = cachedData[1];
  var rowsAnalyzed    = cachedData[2];
  var minutesW1       = cachedData[3];
  var minutesW3       = cachedData[4];
  var minutesW4       = cachedData[5];
  var minutesW5       = cachedData[6];
  var minutesP1       = cachedData[7];
  var minutesP2       = cachedData[8];

  var recipientSheet = ss.getSheetByName("Report Recipients");
  var notifyTargets = "";
  if (recipientSheet && recipientSheet.getLastRow() >= 1) {
    notifyTargets = recipientSheet.getRange(1, 1, recipientSheet.getLastRow(), 1)
                                  .getValues()
                                  .map(function(r) { return r[0] ? r[0].toString().trim() : ""; })
                                  .filter(function(e) { return e.length > 0; })
                                  .join(", ");
  }

  if (!notifyTargets) return;
