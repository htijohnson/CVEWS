#include <WiFiClientSecure.h>
#include <PubSubClient.h>
#include <WebServer.h>
#include <time.h>

// --- WiFi Configuration ---
const char* ssid     = "HTiCenCom";
const char* password = "headtrauma111";

// --- HiveMQ Cloud Configuration ---
const char* mqtt_server   = "91edf0b43131429b9a2d60d852310ada.s1.eu.hivemq.cloud";
const int   mqtt_port     = 8883;
const char* mqtt_user     = "cvewstelemetry1";
const char* mqtt_password = "water123";

// --- MQTT Topics ---
const char* topic_ip     = "plc/esp32/ip";
const char* topic_status = "plc/esp32/status";
const char* topic_data   = "plc/data/raw";

// --- Pins on ESP32-C3 SuperMini ---
#define RX1_PIN 20  // Connect to MAX3232 TXD
#define TX1_PIN 21  // Connect to MAX3232 RXD

WiFiClientSecure espClient;
PubSubClient client(espClient);
WebServer server(80);

// --- Timers ---
unsigned long lastDashboardPublish = 0;
const unsigned long dashboardInterval = 3000; // Heartbeat every 3s

unsigned long lastPollTime = 0;
const unsigned long pollInterval = 1000;      // Send poll query every 1s
const char* pollCommand = "?\r\n";

// --- Global Status Variables ---
String latestPlcRaw = "Waiting for serial data...";
unsigned long lastPlcRxTime = 0;

// Bit registers for I/O modules (8 bits each)
uint8_t slot1_outputs = 0x00; // Y000 - Y007
uint8_t slot2_inputs  = 0x00; // X010 - X017
uint8_t slot3_inputs  = 0x00; // X020 - X027

// --- Parse Received Serial Data ---
void parsePlcData(String payload) {
  latestPlcRaw = payload;
  lastPlcRxTime = millis();

  // Try extracting hex/decimal numbers if formatted as IN=value or comma-separated
  // Example formats handled: "0x12,0x34" or standard integer values
  int inIdx = payload.indexOf("IN=");
  if (inIdx >= 0) {
    slot2_inputs = payload.substring(inIdx + 3).toInt();
  }
  int outIdx = payload.indexOf("OUT=");
  if (outIdx >= 0) {
    slot1_outputs = payload.substring(outIdx + 4).toInt();
  }
}

void connectWiFi() {
  Serial.print("Connecting to WiFi: ");
  Serial.println(ssid);
  WiFi.mode(WIFI_STA);
  WiFi.begin(ssid, password);

  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }

  Serial.println("\nWiFi connected!");
  Serial.print("ESP32 Local IP: ");
  Serial.println(WiFi.localIP());

  configTime(0, 0, "pool.ntp.org");
}

void reconnectMQTT() {
  while (!client.connected()) {
    Serial.print("Attempting MQTT connection to HiveMQ...");
    String clientId = "ESP32_C3_Client-" + String(random(0xffff), HEX);

    if (client.connect(clientId.c_str(), mqtt_user, mqtt_password)) {
      Serial.println(" CONNECTED!");
      
      client.publish(topic_status, "online", true);
      String ipAddress = WiFi.localIP().toString();
      client.publish(topic_ip, ipAddress.c_str(), true);
    } else {
      Serial.print(" failed, rc=");
      Serial.print(client.state());
      Serial.println(" - Retrying in 4s...");
      delay(4000);
    }
  }
}

void setup() {
  Serial.begin(115200);

  // Initialize UART1 for MAX3232 to PLC
  Serial1.begin(9600, SERIAL_8N1, RX1_PIN, TX1_PIN);

  espClient.setInsecure();
  client.setServer(mqtt_server, mqtt_port);

  connectWiFi();

  // Start internal HTTP server
  server.on("/", handleRoot);
  server.begin();
  Serial.println("Local HTTP Web Server active at http://" + WiFi.localIP().toString());
}

void loop() {
  if (WiFi.status() != WL_CONNECTED) {
    connectWiFi();
  }

  if (!client.connected()) {
    reconnectMQTT();
  }
  client.loop();
  server.handleClient(); // Handle local web dashboard visits

  // Heartbeat publish to HiveMQ every 3 seconds
  if (millis() - lastDashboardPublish >= dashboardInterval) {
    lastDashboardPublish = millis();
    String ipStr = WiFi.localIP().toString();
    client.publish(topic_ip, ipStr.c_str(), true);
    client.publish(topic_status, "online", true);
  }

  // Polling query loop
  if (millis() - lastPollTime >= pollInterval) {
    lastPollTime = millis();
    Serial1.print(pollCommand);
  }

  // Read incoming response from PLC
  if (Serial1.available()) {
    String incomingPayload = Serial1.readStringUntil('\n');
    incomingPayload.trim();

    if (incomingPayload.length() > 0) {
      Serial.print("Received from PLC: ");
      Serial.println(incomingPayload);

      // Parse status bits and update local page
      parsePlcData(incomingPayload);

      // Forward to MQTT broker
      client.publish(topic_data, incomingPayload.c_str());
    }
  }
}
