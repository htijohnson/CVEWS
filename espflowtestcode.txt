// RS485 Control Callbacks
void preTransmission() {

float registersToFloat(uint16_t reg0, uint16_t reg1) {
  uint32_t combined = ((uint32_t)reg1 << 16) | reg0; 
  float result;
  memcpy(&result, &combined, sizeof(result));
  return result;
}

float readAnalogSensor(uint8_t pin, float minVal, float maxVal) {
  uint32_t sum = 0;
  for (int i = 0; i < 16; i++) {
    sum += analogRead(pin);
    delay(2);
  }
  float adcAvg = sum / 16.0;
  float scaled = ((adcAvg - 745.0) / (3722.0 - 745.0)) * (maxVal - minVal) + minVal;
  return constrain(scaled, minVal, maxVal);
}

void readModbusData() {
  node.clearResponseBuffer();
  uint8_t flowResult = node.readHoldingRegisters(0x0000, 2);
  float flowRate = 0.0;
  bool flowOk = false;

  if (flowResult == node.ku8MBSuccess) {
    flowRate = registersToFloat(node.getResponseBuffer(0), node.getResponseBuffer(1));
    flowOk = true;
  }

  delay(50);

  node.clearResponseBuffer();
  uint8_t totalResult = node.readHoldingRegisters(0x0008, 2);
  float totalFlow = 0.0;
  bool totalOk = false;

  if (totalResult == node.ku8MBSuccess) {
    totalFlow = registersToFloat(node.getResponseBuffer(0), node.getResponseBuffer(1));
    if (totalFlow < 0.00001f && totalFlow > 0.0f) {
      totalFlow = 0.0f;
    }
    totalOk = true;
  }

  if (flowOk) currentFlowRate = flowRate;
  if (totalOk) currentTotalFlow = totalFlow;

  lastModbusSuccess = flowOk;
}

// Read dry contact on GPIO 3 with debounce
int readAlarmDryContact() {
  if (digitalRead(PIN_ALM) == LOW) {
    delay(20);
    if (digitalRead(PIN_ALM) == LOW) {
      return 1; // Closed to GND -> Alarm Active/Tripped
    }
  }
  return 0;   // Open -> Clear
}

// Generic POST dispatcher to trigger SMS actions in Google Apps Script
void sendScriptSmsAction(const char* actionName, const char* alertText) {
  if (WiFi.status() != WL_CONNECTED) {
    Serial.println("[SMS Alert] WiFi disconnected, unable to trigger SMS.");
    return;
  }

  Serial.printf("[SMS Alert] Dispatching action '%s'...\n", actionName);

  WiFiClientSecure gClient;
  gClient.setInsecure();

  HTTPClient http;
  http.begin(gClient, GOOGLE_SCRIPT_URL);
  http.addHeader("Content-Type", "application/json");
  http.setFollowRedirects(HTTPC_STRICT_FOLLOW_REDIRECTS);
  http.setTimeout(10000);

  StaticJsonDocument<256> alertDoc;
  alertDoc["action"] = actionName;
  alertDoc["message"] = alertText;

  char buffer[256];
  serializeJson(alertDoc, buffer);

  int httpResponseCode = http.POST(buffer);
  Serial.printf("[SMS Alert] Response Code: %d\n", httpResponseCode);
  http.end();
}

// Check alarm duration and trigger SMS if tripped continuously > 30s
void processAlarmTimer() {
  int isAlarmTripped = readAlarmDryContact();

  if (isAlarmTripped == 1) {
    if (alarmActiveStartTime == 0) {
      alarmActiveStartTime = millis();
    } else {
      if (!alarmSmsSent && (millis() - alarmActiveStartTime >= 30000)) {
        sendScriptSmsAction("alarm_sms", "🚨 CVEWS CRITICAL ALERT: The System Alarm Light has been ACTIVE for over 30 seconds! Check telemetry dashboard: https://htijohnson.github.io/CVEWS/");
        alarmSmsSent = true; 
      }
    }
  } else {
    alarmActiveStartTime = 0;
    alarmSmsSent = false;
  }
}

// Evaluates system clock daily and triggers a noon SMS
void checkDailyNoonTest() {
  struct tm timeinfo;
  if (!getLocalTime(&timeinfo)) {
    return; // Clock not yet synced with NTP
  }

  // Trigger test if hour == 12 (12:00 PM) and hasn't run yet today
  if (timeinfo.tm_hour == 12 && lastNoonTestDay != timeinfo.tm_mday) {
    Serial.println("[Daily Test] It is 12:00 PM PST. Triggering daily test SMS...");
    
    char dateBuf[64];
    strftime(dateBuf, sizeof(dateBuf), "%m/%d/%Y at %I:%M %p", &timeinfo);
    
    String msg = "CVEWS DAILY SYSTEM TEST: Telemetry system operational. Local time: " + String(dateBuf) + ". Dashboard: https://htijohnson.github.io/CVEWS/";
    sendScriptSmsAction("daily_test_sms", msg.c_str());
    
    lastNoonTestDay = timeinfo.tm_mday; // Latch to today's date
  }
}

StaticJsonDocument<384> buildTelemetryPayload() {
  StaticJsonDocument<384> doc;

  doc["tk_sm"] = readAnalogSensor(PIN_TK_SM, 0.0, 4100.0);
  doc["tk_lg"] = readAnalogSensor(PIN_TK_LG, 0.0, 71000.0);
  doc["cl2"]   = readAnalogSensor(PIN_CL2, 0.0, 2.00);
  
  float gpm = (currentFlowRate * 264.172) / 60.0;
  doc["flow"]  = lastModbusSuccess ? gpm : 0.0;

  // Digital Inputs
  doc["w1"]  = 0;
  doc["w3"]  = (digitalRead(PIN_W3) == LOW) ? 1 : 0;
  doc["w4"]  = (digitalRead(PIN_W4) == LOW) ? 1 : 0;
  doc["w5"]  = (digitalRead(PIN_W5) == LOW) ? 1 : 0;
  doc["p1"]  = (digitalRead(PIN_P1) == LOW) ? 1 : 0;
  doc["p2"]  = (digitalRead(PIN_P2) == LOW) ? 1 : 0;

  // Alarm Light Dry Contact (1 = Tripped, 0 = Clear)
  doc["alm"] = readAlarmDryContact();

  doc["ip"]      = WiFi.localIP().toString();
  doc["version"] = current_firmware_version;

  return doc;
}

// Encrypted GitHub OTA Engine
bool checkAndPerformGithubOTA() {
  Serial.println("[OTA] Checking GitHub for firmware updates...");
  mqttClient.publish(MQTT_TOPIC_STATUS, "ota_checking");

  WiFiClientSecure otaClient;
  otaClient.setInsecure();

  HTTPClient http;
  http.setFollowRedirects(HTTPC_STRICT_FOLLOW_REDIRECTS);

  if (!http.begin(otaClient, github_version_url)) {
    Serial.println("[OTA] Failed to connect to version URL.");
    mqttClient.publish(MQTT_TOPIC_STATUS, "ota_failed_conn");
    return false;
  }

  int httpCode = http.GET();
  if (httpCode != HTTP_CODE_OK) {
    Serial.printf("[OTA] Version HTTP failed: %d\n", httpCode);
    http.end();
    mqttClient.publish(MQTT_TOPIC_STATUS, "ota_failed_version_http");
    return false;
  }

  String remoteVersion = http.getString();
  remoteVersion.trim();
  remoteVersion.replace("\r", "");
  remoteVersion.replace("\n", "");
  http.end();

  if (remoteVersion.equals(String(current_firmware_version))) {
    Serial.println("[OTA] Firmware is already up to date.");
    mqttClient.publish(MQTT_TOPIC_STATUS, "ota_up_to_date");
    return false;
  }

  Serial.println("[OTA] New version found! Downloading encrypted binary...");
  mqttClient.publish(MQTT_TOPIC_STATUS, "ota_downloading");

  if (!http.begin(otaClient, github_firmware_url)) {
    Serial.println("[OTA] Failed to connect to binary URL.");
    mqttClient.publish(MQTT_TOPIC_STATUS, "ota_failed_bin_conn");
    return false;
  }

  httpCode = http.GET();
  if (httpCode != HTTP_CODE_OK) {
    Serial.printf("[OTA] Binary download failed: %d\n", httpCode);
    http.end();
    mqttClient.publish(MQTT_TOPIC_STATUS, "ota_failed_bin_http");
    return false;
  }

  int contentLength = http.getSize();
  if (contentLength <= 0 || !Update.begin(contentLength)) {
    Serial.println("[OTA] Partition space allocation error.");
    http.end();
    mqttClient.publish(MQTT_TOPIC_STATUS, "ota_failed_space");
    return false;
  }

  mbedtls_aes_context aes;
  mbedtls_aes_init(&aes);
  mbedtls_aes_setkey_dec(&aes, aes_key, 256);

  uint8_t iv_buffer[16];
  memcpy(iv_buffer, aes_iv, 16);

  WiFiClient* stream = http.getStreamPtr();
  const size_t CHUNK_SIZE = 512;
  uint8_t encryptedBuffer[CHUNK_SIZE];
  uint8_t decryptedBuffer[CHUNK_SIZE];

  size_t totalBytesWritten = 0;

  while (http.connected() && (totalBytesWritten < contentLength)) {
    size_t availableBytes = stream->available();
    if (availableBytes > 0) {
      size_t readLen = stream->readBytes(encryptedBuffer, min(availableBytes, CHUNK_SIZE));
      size_t blocksLen = (readLen / 16) * 16;

      if (blocksLen > 0) {
        mbedtls_aes_crypt_cbc(&aes, MBEDTLS_AES_DECRYPT, blocksLen, iv_buffer, encryptedBuffer, decryptedBuffer);
        size_t written = Update.write(decryptedBuffer, blocksLen);

        if (written != blocksLen) {
          mbedtls_aes_free(&aes);
          http.end();
          mqttClient.publish(MQTT_TOPIC_STATUS, "ota_flash_error");
          return false;
        }

        totalBytesWritten += blocksLen;
      }
    }
    delay(1);
  }

  mbedtls_aes_free(&aes);
  http.end();

  if (Update.end(true)) {
    Serial.println("[OTA] Update successful! Rebooting...");
    mqttClient.publish(MQTT_TOPIC_STATUS, "ota_success_rebooting");
    delay(1000);
    return true;
  } else {
    mqttClient.publish(MQTT_TOPIC_STATUS, "ota_failed_finalize");
    return false;
  }
}

// MQTT Listener
void mqttCallback(char* topic, byte* payload, unsigned int length) {
  String message = "";
  for (unsigned int i = 0; i < length; i++) {
    message += (char)payload[i];
  }
  message.trim();

  if (String(topic) == MQTT_TOPIC_CMD) {
    if (message.equalsIgnoreCase("UPDATE") || message.equalsIgnoreCase("OTA") || message.equalsIgnoreCase("CHECK")) {
      otaRequestedViaMqtt = true;
    }
  }
}

}

void reconnectMqtt() {
  while (!mqttClient.connected()) {
    String clientId = "ESP32_CVEWS_" + String(random(0xffff), HEX);
    Serial.print("[MQTT] Connecting to HiveMQ...");
    
    if (mqttClient.connect(clientId.c_str(), MQTT_USER, MQTT_PASS, MQTT_TOPIC_STATUS, 1, true, "offline")) {
      Serial.println(" Connected!");
      mqttClient.publish(MQTT_TOPIC_STATUS, "online", true);
      mqttClient.subscribe(MQTT_TOPIC_CMD);
      break;
    } else {
      Serial.print(" Failed, rc=");
      Serial.print(mqttClient.state());
      Serial.println(" Retrying in 3 seconds...");
      delay(3000);
    }
  }
}

void setup() {
  Serial.begin(115200);
  delay(1000);

  pinMode(LED_PIN, OUTPUT);
  pinMode(MAX485_DE, OUTPUT);
  digitalWrite(MAX485_DE, LOW);

  pinMode(PIN_P1, INPUT_PULLUP);
  pinMode(PIN_W5, INPUT_PULLUP);
  pinMode(PIN_W4, INPUT_PULLUP);
  pinMode(PIN_W3, INPUT_PULLUP);
  pinMode(PIN_ALM, INPUT_PULLUP); // Dry contact on GPIO 3
  pinMode(PIN_P2, INPUT_PULLUP);

  analogReadResolution(12);

  // RS485 Modbus Serial
  Serial1.begin(9600, SERIAL_8N1, RX_PIN, TX_PIN);
  node.begin(1, Serial1);
  node.preTransmission(preTransmission);
  node.postTransmission(postTransmission);

  setupWiFi();

  mqttClient.setServer(MQTT_SERVER, MQTT_PORT);
  mqttClient.setCallback(mqttCallback);

  server.on("/", HTTP_GET, handleRoot);
  server.on("/update", HTTP_GET, handleUpdatePage);
  server.on("/update-github", HTTP_GET, handleGithubOtaTrigger);

  server.on("/update", HTTP_POST, []() {
    server.sendHeader("Connection", "close");
    server.send(200, "text/html", "<body style='background:#1a1e24;color:#e2e8f0;font-family:sans-serif;padding:30px;'><h2>OTA Complete!</h2><p>Rebooting... Please wait 10 seconds.</p><script>setTimeout(()=>{window.location.href='/';}, 10000);</script></body>");
    ESP.restart();
  }, []() {
    HTTPUpload& upload = server.upload();
    if (upload.status == UPLOAD_FILE_START) {
      if (!Update.begin(UPDATE_SIZE_UNKNOWN)) {
        Update.printError(Serial);
      }
    } else if (upload.status == UPLOAD_FILE_WRITE) {
      if (Update.write(upload.buf, upload.currentSize) != upload.currentSize) {
        Update.printError(Serial);
      }
    } else if (upload.status == UPLOAD_FILE_END) {
      if (Update.end(true)) {
        Serial.printf("[OTA] Local Update Success: %u Bytes\n", upload.totalSize);
      } else {
        Update.printError(Serial);
      }
    }
  });

  server.begin();
}

void loop() {
  if (WiFi.status() != WL_CONNECTED) setupWiFi();
  if (!mqttClient.connected()) reconnectMqtt();
  mqttClient.loop();
  
  server.handleClient();

  // Evaluate 30-second alarm duration timer
  processAlarmTimer();

  // Evaluate clock for daily noon test SMS
  checkDailyNoonTest();

  if (otaRequestedViaMqtt) {
    otaRequestedViaMqtt = false;
    if (checkAndPerformGithubOTA()) {
      ESP.restart();
    }
  }

  unsigned long currentMillis = millis();

  // Status LED blinker
  if (currentMillis - previousMillis >= blinkInterval) {
    previousMillis = currentMillis;
    ledState = !ledState;
    digitalWrite(LED_PIN, ledState);
  }

  // Publish to MQTT every 3 seconds
  if (currentMillis - lastMqttPub >= 3000) {
    lastMqttPub = currentMillis;
    readModbusData();
    
    StaticJsonDocument<384> doc = buildTelemetryPayload();
    char buffer[384];
    serializeJson(doc, buffer);
    
    mqttClient.publish(MQTT_TOPIC, buffer);
  }

  // Post to Google Sheets logger every 30 seconds
  if (currentMillis - lastGScriptPost >= 30000) {
    lastGScriptPost = currentMillis;
    StaticJsonDocument<384> doc = buildTelemetryPayload();
    char buffer[384];
    serializeJson(doc, buffer);
    
    sendGoogleSheetData(String(buffer));
  }

  // Periodic GitHub OTA Check (every 12 hours)
  if (currentMillis - lastOtaCheck >= 43200000) {
    lastOtaCheck = currentMillis;
    if (checkAndPerformGithubOTA()) {
      ESP.restart();
    }
  }
}
